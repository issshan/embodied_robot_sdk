cmake_minimum_required(VERSION 3.15)
project(embodied_robot_sdk VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(robot_sdk src/ring_buffer.cpp src/protocol_parser.cpp src/tcp_client.cpp src/robot_sdk.cpp)
target_include_directories(robot_sdk PUBLIC include)
if(MSVC)
  target_compile_options(robot_sdk PRIVATE /W4)
else()
  target_compile_options(robot_sdk PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_compile_definitions(robot_sdk PUBLIC _WIN32_WINNT=0x0600 NOMINMAX)
  target_link_libraries(robot_sdk PUBLIC ws2_32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(robot_sdk PUBLIC Threads::Threads)
endif()
add_executable(robot_demo examples/robot_demo.cpp)
target_link_libraries(robot_demo PRIVATE robot_sdk)
include(CTest)
if(BUILD_TESTING)
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE robot_sdk)
  add_test(NAME protocol COMMAND protocol_tests)
  add_executable(integration_client tests/integration_client.cpp)
  target_link_libraries(integration_client PRIVATE robot_sdk)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py $<TARGET_FILE:integration_client>)
    set_tests_properties(integration PROPERTIES TIMEOUT 60)
  endif()
endif()
